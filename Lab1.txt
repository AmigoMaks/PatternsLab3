using System;
using System.Collections.Generic;

namespace BehavioralPatternsDemo
{
    // 1. CHAIN OF RESPONSIBILITY
    public abstract class Officer
    {
        protected Officer _nextOfficer;
        public void SetNext(Officer next) => _nextOfficer = next;
        public abstract void HandleSignal(int dangerLevel);
    }

    public class Lieutenant : Officer
    {
        public override void HandleSignal(int dangerLevel)
        {
            if (dangerLevel < 5) Console.WriteLine("[Chain] Лейтенант розiбрався з дрiбною загрозою");
            else if (_nextOfficer != null) _nextOfficer.HandleSignal(dangerLevel);
        }
    }

    public class Admiral : Officer
    {
        public override void HandleSignal(int dangerLevel) =>
            Console.WriteLine("[Chain] Адмiрал приймає командування! Загроза критична");
    }

    // 2. COMMAND
    public interface ICommand { void Execute(); }

    public class MoveCommand : ICommand
    {
        private string _unit;
        public MoveCommand(string unit) { _unit = unit; }
        public void Execute() => Console.WriteLine($"[Command] {_unit} перемiщується");
    }

    public class AttackCommand : ICommand
    {
        private string _unit;
        public AttackCommand(string unit) { _unit = unit; }
        public void Execute() => Console.WriteLine($"[Command] {_unit} атакує цiль");
    }

    public class CommandQueue
    {
        private Queue<ICommand> _queue = new Queue<ICommand>();
        public void AddCommand(ICommand cmd) => _queue.Enqueue(cmd);
        public void ExecuteAll() { while (_queue.Count > 0) _queue.Dequeue().Execute(); }
    }

    // 3. ITERATOR
    public class Army : System.Collections.IEnumerable
    {
        private List<string> _units = new List<string> { "Танк", "Морпiх", "Дрон" };
        public System.Collections.IEnumerator GetEnumerator() => _units.GetEnumerator();
    }

    // 4. MEDIATOR
    public interface ISpaceControl { void RequestLanding(string shipName); }

    public class SpaceportControl : ISpaceControl
    {
        public void RequestLanding(string shipName) =>
            Console.WriteLine($"[Mediator] Диспетчер: Корабель {shipName}, посадковий майданчик вiльний");
    }

    public class TransportShip
    {
        private ISpaceControl _control;
        private string _name;
        public TransportShip(string name, ISpaceControl control) { _name = name; _control = control; }
        public void Land() => _control.RequestLanding(_name);
    }

    // 5. MEMENTO
    public class GameSave { public int Score { get; } public GameSave(int score) { Score = score; } }

    public class GameEngine
    {
        public int Score { get; set; }
        public GameSave Save() => new GameSave(Score);
        public void Load(GameSave save) => Score = save.Score;
    }

    // 6. OBSERVER
    public interface IObserver { void Update(string eventInfo); }

    public class AutoTurret : IObserver
    {
        public void Update(string eventInfo) =>
            Console.WriteLine($"[Observer] Турель отримала координати: {eventInfo}. Вiдкриваю вогонь");
    }

    public class RadarSystem
    {
        private List<IObserver> _turrets = new List<IObserver>();
        public void Attach(IObserver observer) => _turrets.Add(observer);
        public void DetectEnemy()
        {
            Console.WriteLine("[Observer] Радар: Виявлено ворога у секторi 7");
            foreach (var t in _turrets) t.Update("Сектор 7");
        }
    }

    // 7. STATE
    public interface IUnitState { void HandleBehavior(); }
    public class IdleState : IUnitState { public void HandleBehavior() => Console.WriteLine("[State] Юнiт стоїть на мiсцi"); }
    public class AttackState : IUnitState { public void HandleBehavior() => Console.WriteLine("[State] Юнiт стрiляє у ворога"); }

    public class SmartUnit
    {
        private IUnitState _state = new IdleState();
        public void SetState(IUnitState state) => _state = state;
        public void Act() => _state.HandleBehavior();
    }

    // 8. STRATEGY
    public interface ITactic { void ExecuteTactic(); }
    public class StealthTactic : ITactic { public void ExecuteTactic() => Console.WriteLine("[Strategy] Режим маскування активовано"); }
    public class BlitzkriegTactic : ITactic { public void ExecuteTactic() => Console.WriteLine("[Strategy] Атака всiма силами"); }

    public class Commander
    {
        private ITactic _tactic;
        public Commander(ITactic tactic) { _tactic = tactic; }
        public void ExecutePlan() => _tactic.ExecuteTactic();
    }

    // 9. TEMPLATE METHOD
    public abstract class ConstructionRoutine
    {
        public void Build()
        {
            LayFoundation();
            ErectWalls();
            Console.WriteLine("[Template] Будiвництво завершено");
        }
        protected abstract void LayFoundation();
        protected abstract void ErectWalls();
    }

    public class HumanBarracks : ConstructionRoutine
    {
        protected override void LayFoundation() => Console.WriteLine("[Template] Заливка бетону...");
        protected override void ErectWalls() => Console.WriteLine("[Template] Встановлення сталевих стiн...");
    }

    // 10. VISITOR
    public interface IVisitor { void Visit(Mech mech); void Visit(Base structure); }

    public class Mechanic : IVisitor
    {
        public void Visit(Mech mech) => Console.WriteLine("[Visitor] Механiк лагодить сервоприводи Меха");
        public void Visit(Base structure) => Console.WriteLine("[Visitor] Механiк вiдновлює енергощити Бази");
    }

    public interface IVisitable { void Accept(IVisitor visitor); }
    public class Mech : IVisitable { public void Accept(IVisitor visitor) => visitor.Visit(this); }
    public class Base : IVisitable { public void Accept(IVisitor visitor) => visitor.Visit(this); }

    class Program
    {
        static void Main()
        {
            // Chain of Responsibility
            Officer lt = new Lieutenant();
            Officer admiral = new Admiral();
            lt.SetNext(admiral);
            lt.HandleSignal(10);

            // Command
            CommandQueue queue = new CommandQueue();
            queue.AddCommand(new MoveCommand("Загiн Альфа"));
            queue.AddCommand(new AttackCommand("Загiн Альфа"));
            queue.ExecuteAll();

            // Iterator
            Army myArmy = new Army();
            Console.Write("[Iterator] Перекличка армiї: ");
            foreach (string unit in myArmy) Console.Write($"{unit} ");
            Console.WriteLine();

            // Mediator
            SpaceportControl tower = new SpaceportControl();
            TransportShip ship = new TransportShip("Аполлон", tower);
            ship.Land();

            // Memento
            GameEngine game = new GameEngine() { Score = 1500 };
            GameSave quickSave = game.Save();
            game.Score = 0;
            game.Load(quickSave);
            Console.WriteLine($"[Memento] Вiдновлено рахунок: {game.Score}");

            // Observer
            RadarSystem radar = new RadarSystem();
            radar.Attach(new AutoTurret());
            radar.DetectEnemy();

            // State
            SmartUnit bot = new SmartUnit();
            bot.Act();
            bot.SetState(new AttackState());
            bot.Act();

            // Strategy
            Commander cmdr = new Commander(new StealthTactic());
            cmdr.ExecutePlan();

            // Template Method
            ConstructionRoutine humanBarracks = new HumanBarracks();
            humanBarracks.Build();

            //  Visitor
            Mechanic eng = new Mechanic();
            IVisitable myMech = new Mech();
            myMech.Accept(eng);
        }
    }
}
